# Examples for libbolza. Included from the bolza project CMakeLists.txt.
# Builds terminal OCP demos. No autolink, automsgs, or autonomy.

add_executable(bolza_robot "${CMAKE_CURRENT_LIST_DIR}/robot/solve_robot.cpp")
target_link_libraries(bolza_robot PRIVATE bolza)
target_include_directories(bolza_robot PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
set_target_properties(bolza_robot PROPERTIES
  CXX_STANDARD 20
  CXX_STANDARD_REQUIRED ON
  CXX_EXTENSIONS OFF)

include("${CMAKE_CURRENT_LIST_DIR}/robot/ballbot/CMakelists.txt")

install(TARGETS bolza_robot RUNTIME DESTINATION bin)

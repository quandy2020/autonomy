# Ballbot terminal OCP. Included from examples/CMakelists.txt.
# Links libbolza and glog.

find_package(GLog REQUIRED)

add_executable(bolza_ballbot
  "${CMAKE_CURRENT_LIST_DIR}/ballbot.cpp"
  "${CMAKE_CURRENT_LIST_DIR}/main.cpp")
target_link_libraries(bolza_ballbot PRIVATE bolza GLog::GLog)
get_filename_component(BOLZA_EXAMPLES_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
target_include_directories(bolza_ballbot PRIVATE "${BOLZA_EXAMPLES_DIR}")
set_target_properties(bolza_ballbot PROPERTIES
  CXX_STANDARD 20
  CXX_STANDARD_REQUIRED ON
  CXX_EXTENSIONS OFF)

install(TARGETS bolza_ballbot RUNTIME DESTINATION bin)
